cmake_minimum_required(VERSION 3.25)
project(poker_evaluator VERSION 0.1.0 LANGUAGES C CXX)

option(POKER_BUILD_TESTS "Build correctness tests" ${PROJECT_IS_TOP_LEVEL})
option(POKER_BUILD_BENCHMARKS "Build the trusted measurement driver" ${PROJECT_IS_TOP_LEVEL})
option(POKER_NATIVE "Compile for the build machine (not portable binaries)" OFF)
option(POKER_LTO "Enable interprocedural optimization" OFF)
option(POKER_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(POKER_TSAN "Enable thread sanitizer (separate build)" OFF)
option(POKER_FUZZ "Build the Clang libFuzzer target" OFF)
set(POKER_EXHAUSTIVE_MAX_CARDS 5 CACHE STRING "Largest exhaustive domain registered with CTest (5, 6, or 7)")
if(NOT POKER_EXHAUSTIVE_MAX_CARDS MATCHES "^[567]$")
  message(FATAL_ERROR "POKER_EXHAUSTIVE_MAX_CARDS must be 5, 6, or 7")
endif()
if(POKER_SANITIZE AND POKER_TSAN)
  message(FATAL_ERROR "ASan and TSan require separate builds")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(POKER_LTO)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT ipo_ok OUTPUT ipo_error)
  if(NOT ipo_ok)
    message(FATAL_ERROR "IPO unavailable: ${ipo_error}")
  endif()
  set(CMAKE_INTERPROCEDURAL_OPTIMIZATION ON)
endif()
if(POKER_NATIVE)
  if(MSVC)
    message(FATAL_ERROR "Select an explicit MSVC architecture profile instead of POKER_NATIVE")
  elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
    add_compile_options(-mcpu=native)
  else()
    add_compile_options(-march=native)
  endif()
endif()
if(POKER_SANITIZE OR POKER_TSAN)
  if(MSVC)
    message(FATAL_ERROR "These sanitizer presets require GCC or Clang")
  endif()
  if(POKER_SANITIZE)
    set(poker_sanitizers address,undefined)
  else()
    set(poker_sanitizers thread)
  endif()
  add_compile_options(-fsanitize=${poker_sanitizers} -fno-omit-frame-pointer)
  add_link_options(-fsanitize=${poker_sanitizers})
endif()

# Optimization candidates may change these sources, never this build policy.
file(GLOB_RECURSE poker_sources CONFIGURE_DEPENDS src/*.cpp)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND NOT MSVC)
  enable_language(ASM)
  file(GLOB_RECURSE poker_assembly CONFIGURE_DEPENDS src/*.S)
  list(APPEND poker_sources ${poker_assembly})
endif()
add_library(poker_evaluator ${poker_sources})
add_library(poker::evaluator ALIAS poker_evaluator)
target_include_directories(poker_evaluator PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_compile_features(poker_evaluator PUBLIC cxx_std_20)
set_target_properties(poker_evaluator PROPERTIES EXPORT_NAME evaluator POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(poker_evaluator PRIVATE /W4)
else()
  target_compile_options(poker_evaluator PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
install(TARGETS poker_evaluator EXPORT pokerTargets
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT pokerTargets NAMESPACE poker:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/poker)
configure_package_config_file(cmake/pokerConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/pokerConfig.cmake
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/poker)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/pokerConfigVersion.cmake
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/pokerConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/pokerConfigVersion.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/poker)

if(POKER_BUILD_TESTS OR POKER_BUILD_BENCHMARKS)
  include(FetchContent)
  # Immutable upstream commits: GoogleTest v1.17.0 and Benchmark v1.9.4.
  set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
  set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
  set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
  FetchContent_Declare(googletest GIT_REPOSITORY https://github.com/google/googletest.git
    GIT_TAG 52eb8108c5bdec04579160ae17225d66034bd723)
  set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  set(BENCHMARK_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(googlebenchmark GIT_REPOSITORY https://github.com/google/benchmark.git
    GIT_TAG 31c66f47f60d06790a4885a261d682ba339bd100)
  add_library(poker_oracle STATIC tests/oracle.cpp)
  target_include_directories(poker_oracle PUBLIC tests)
endif()
if(POKER_BUILD_TESTS)
  enable_testing()
  FetchContent_MakeAvailable(googletest)
  find_package(Threads REQUIRED)
  add_executable(poker_tests tests/tests.cpp)
  target_link_libraries(poker_tests PRIVATE poker::evaluator poker_oracle GTest::gtest_main Threads::Threads)
  add_test(NAME unit COMMAND poker_tests)
  set_tests_properties(unit PROPERTIES LABELS "quick;correctness" TIMEOUT 180)
  if(POKER_SANITIZE OR POKER_TSAN)
    # Instrumented reference enumeration takes over three minutes on hosted
    # runners. Keep every case and allow sanitizer builds enough time to finish.
    set_tests_properties(unit PROPERTIES TIMEOUT 600)
  endif()
  add_executable(poker_c_consumer tests/c_consumer.c)
  target_link_libraries(poker_c_consumer PRIVATE poker::evaluator)
  set_target_properties(poker_c_consumer PROPERTIES LINKER_LANGUAGE CXX C_STANDARD 11)
  add_test(NAME c_consumer COMMAND poker_c_consumer)
  set_tests_properties(c_consumer PROPERTIES LABELS "quick;integration")
  # Installed consumers use an ordinary release build, independently of this
  # build's sanitizer/runtime flags. Sanitized archives need matching consumers.
  if(NOT POKER_SANITIZE AND NOT POKER_TSAN AND NOT POKER_FUZZ)
    add_test(NAME install_consumer COMMAND ${CMAKE_COMMAND}
      -DBUILD_DIR=${CMAKE_CURRENT_BINARY_DIR} -DSOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}
      -DCONFIG=$<CONFIG> -P ${CMAKE_CURRENT_SOURCE_DIR}/tools/check_install.cmake)
    set_tests_properties(install_consumer PROPERTIES LABELS "quick;integration" RUN_SERIAL TRUE TIMEOUT 180)
  endif()
  add_executable(poker_exhaustive tests/exhaustive.cpp)
  target_link_libraries(poker_exhaustive PRIVATE poker::evaluator poker_oracle)
  foreach(cards RANGE 5 ${POKER_EXHAUSTIVE_MAX_CARDS})
    foreach(shard RANGE 0 7)
      add_test(NAME exhaustive_${cards}_${shard} COMMAND poker_exhaustive ${cards} ${shard} 8)
      set_tests_properties(exhaustive_${cards}_${shard} PROPERTIES LABELS "exhaustive;correctness" TIMEOUT 14400)
    endforeach()
  endforeach()
  find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
  add_test(NAME gate_tests COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tools -p "test_*.py")
  set_tests_properties(gate_tests PROPERTIES LABELS "quick;gate" TIMEOUT 120)
endif()
if(POKER_BUILD_BENCHMARKS)
  FetchContent_MakeAvailable(googlebenchmark)
  add_executable(poker_bench benchmarks/benchmarks.cpp)
  target_link_libraries(poker_bench PRIVATE poker::evaluator poker_oracle benchmark::benchmark)
  if(POKER_BUILD_TESTS)
    add_test(NAME benchmark_smoke COMMAND poker_bench --smoke --seed=12345)
    set_tests_properties(benchmark_smoke PROPERTIES LABELS "benchmark;smoke" RUN_SERIAL TRUE TIMEOUT 180)
  endif()
endif()
if(POKER_FUZZ)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "POKER_FUZZ requires Clang")
  endif()
  if(NOT TARGET poker_oracle)
    add_library(poker_oracle STATIC tests/oracle.cpp)
    target_include_directories(poker_oracle PUBLIC tests)
  endif()
  add_executable(poker_fuzz tests/fuzz.cpp)
  target_compile_options(poker_evaluator PRIVATE -fsanitize=fuzzer-no-link)
  target_compile_options(poker_oracle PRIVATE -fsanitize=fuzzer-no-link)
  target_link_libraries(poker_fuzz PRIVATE poker::evaluator poker_oracle)
  target_compile_options(poker_fuzz PRIVATE -fsanitize=fuzzer)
  target_link_options(poker_fuzz PRIVATE -fsanitize=fuzzer)
endif()
