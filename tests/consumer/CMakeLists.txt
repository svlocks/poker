cmake_minimum_required(VERSION 3.25)
project(poker_installed_consumer LANGUAGES C CXX)
find_package(poker 0.1 CONFIG REQUIRED)
add_executable(c_consumer ../c_consumer.c)
target_link_libraries(c_consumer PRIVATE poker::evaluator)
set_target_properties(c_consumer PROPERTIES LINKER_LANGUAGE CXX C_STANDARD 11)
add_executable(cpp_consumer main.cpp)
target_link_libraries(cpp_consumer PRIVATE poker::evaluator)
enable_testing()
add_test(NAME installed_c COMMAND c_consumer)
add_test(NAME installed_cpp COMMAND cpp_consumer)
